function renderAdminSeatBusOptions() { const sel = $('adminSeatBus'); if (!sel) return; const keep = sel.value; // 수정: 예약 상태나 차량 종류 제한을 풀고 45인승/28인승 버스는 모두 표시하거나, 전체 버스를 대상으로 표시 const active = resRows.filter(r => ['bus45','bus28'].includes(r.buses?.vehicle_type)); const map = new Map(); active.forEach(r => { const b = r.buses; if (b?.id) map.set(String(b.id), `${b.trips?.trip_date ? shortDate(b.trips.trip_date) + ' · ' : ''}${tripName(b.trips?.clubs?.name, b.trips?.title)} · ${b.bus_no}호차`); }); // 만약 resRows에 아직 데이터가 없다면 adminTrips에 등록된 버스 목록을 활용할 수도 있습니다. if (map.size === 0 && typeof adminTrips !== 'undefined') { adminTrips.forEach(t => { (t.buses || []).forEach(b => { if (['bus45','bus28'].includes(b.vehicle_type)) { map.set(String(b.id), `${shortDate(t.trip_date)} · ${tripName(t.clubs?.name, t.title)} · ${b.bus_no}호차`); } }); }); } sel.innerHTML = '' + [...map].map(([id,label]) => ``).join(''); if (map.has(keep)) sel.value = keep; else if (map.size === 1) sel.value = [...map.keys()][0]; if (sel.value) loadAdminSeatBoard(); }
